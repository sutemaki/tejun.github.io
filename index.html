Sub 指定行のE列を選択() Dim wsList As Worksheet Dim wsTarget As Worksheet Dim c As Range Dim rng As Range Set wsList = ThisWorkbook.ActiveSheet Set wsTarget = Workbooks("26年度申請台帳.xlsx").Worksheets("書出し設定申請") For Each c In wsList.Range("A1:A319") If IsNumeric(c.Value) And c.Value <> "" Then If rng Is Nothing Then Set rng = wsTarget.Cells(CLng(c.Value), "E") Else Set rng = Union(rng, wsTarget.Cells(CLng(c.Value), "E")) End If End If Next c If Not rng Is Nothing Then wsTarget.Activate rng.Select End If End Sub
