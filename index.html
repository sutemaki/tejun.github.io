Option Explicit Private Const SRC_BOOK As String = "26年度申請台帳.xlsx" Private Const SRC_SHEET As String = "書出し設定申請" Private Const FILTER_ROW As Long = 3 Private Const FIRST_DATA_ROW As Long = 5 Private gCurrentStep As String '========================================================== ' メイン '========================================================== Public Sub 書出し設定申請_作業ブック作成() Dim srcWb As Workbook Dim srcWs As Worksheet Dim wb As Workbook Dim ws As Worksheet Dim tmpWs As Worksheet Dim summaryWs As Worksheet Dim lastRow As Long Dim workDate As Date Dim inputDate As Variant Dim tempPath As String Dim tempName As String Dim i As Long Dim oldCalc As XlCalculation Dim oldScreen As Boolean Dim oldEvents As Boolean Dim oldAlerts As Boolean Dim oldStatusBar As Variant Dim stateSaved As Boolean '================================================== ' 作業日の入力 '========================================================== inputDate = Application.InputBox( _ Prompt:="作業日を入力してください。", _ Title:="作業日の入力", _ Default:=Format(Date, "yyyy/mm/dd"), _ Type:=2) If VarType(inputDate) = vbBoolean Then If inputDate = False Then Exit Sub End If If Not IsDate(inputDate) Then MsgBox _ "日付を正しく入力してください。", _ vbExclamation Exit Sub End If workDate = DateValue(CDate(inputDate)) '================================================== ' 元ブック確認 '========================================================== On Error Resume Next Set srcWb = Workbooks(SRC_BOOK) On Error GoTo 0 If srcWb Is Nothing Then MsgBox _ SRC_BOOK & " を開いてから実行してください。", _ vbExclamation Exit Sub End If On Error Resume Next Set srcWs = srcWb.Worksheets(SRC_SHEET) On Error GoTo 0 If srcWs Is Nothing Then MsgBox _ "シート「" & SRC_SHEET & "」が見つかりません。", _ vbExclamation Exit Sub End If '================================================== ' Excel状態保存 '========================================================== oldCalc = Application.Calculation oldScreen = Application.ScreenUpdating oldEvents = Application.EnableEvents oldAlerts = Application.DisplayAlerts oldStatusBar = Application.StatusBar stateSaved = True Application.ScreenUpdating = False Application.EnableEvents = False Application.DisplayAlerts = False Application.Calculation = xlCalculationManual On Error GoTo ErrorHandler '================================================== ' 0. ' 元ブック全体をTEMPへ複製 '========================================================== SetStatus "元ブックから作業用コピーを作成" tempName = _ "書出し設定申請_作業用_" & _ Format(Now, "yyyymmdd_hhnnss") & ".xlsx" tempPath = Environ$("TEMP") & "\" & tempName srcWb.SaveCopyAs tempPath '================================================== ' TEMPコピーを開く '========================================================== SetStatus "作業用ブックを開く" Set wb = Workbooks.Open( _ Filename:=tempPath, _ UpdateLinks:=False, _ ReadOnly:=False) Set ws = wb.Worksheets(SRC_SHEET) '================================================== ' 書出し設定申請以外のシート削除 '========================================================== SetStatus "不要シートを削除" For i = wb.Sheets.Count To 1 Step -1 If wb.Sheets(i).Name <> SRC_SHEET Then wb.Sheets(i).Delete End If Next i ClearFilter ws '================================================== ' 2. D列「否認」を削除 '========================================================== SetStatus "D列「否認」の削除" DeleteRowsMatching _ ws, _ 4, _ Array("否認"), _ True '================================================== ' 3. S列空欄へ作業日 '========================================================== SetStatus "S列へ作業日を入力" FillWorkDate ws, workDate '================================================== ' 4. 項番のみの行削除 '========================================================== SetStatus "項番のみの行を削除" DeleteItemOnlyRows ws '================================================== ' 5. ' N列：昇順 ' S列：新しい順 '========================================================== SetStatus "N列・S列の並べ替え" lastRow = LastUsedRow(ws) If lastRow >= FIRST_DATA_ROW Then ws.Range( _ ws.Cells(FIRST_DATA_ROW, 1), _ ws.Cells(lastRow, 26) _ ).Sort _ Key1:=ws.Cells(FIRST_DATA_ROW, 14), _ Order1:=xlAscending, _ Key2:=ws.Cells(FIRST_DATA_ROW, 19), _ Order2:=xlDescending, _ Header:=xlNo, _ MatchCase:=False, _ Orientation:=xlTopToBottom, _ DataOption1:=xlSortTextAsNumbers, _ DataOption2:=xlSortNormal End If '================================================== ' 6. V～Z削除 '========================================================== SetStatus "V～Z列の削除" ws.Columns("V:Z").Delete '================================================== ' 8. 仮想端末 '========================================================== SetStatus "仮想端末シート作成" CreateFilteredSheet _ ws, _ wb, _ 14, _ Array("0", "-"), _ "仮想端末" '================================================== ' 9. 元シートから0/-削除 '========================================================== SetStatus "元シートからN列「0」「-」を削除" DeleteRowsMatching _ ws, _ 14, _ Array("0", "-"), _ True '================================================== ' 7・10相当 ' 同じIT資産No.は最新1件だけ残す '========================================================== SetStatus "IT資産No.の重複データを整理" lastRow = LastUsedRow(ws) If lastRow >= FIRST_DATA_ROW Then ws.Range( _ ws.Cells(FIRST_DATA_ROW, 1), _ ws.Cells(lastRow, 21) _ ).RemoveDuplicates _ Columns:=14, _ Header:=xlNo End If '================================================== ' 11. アンインストール '========================================================== SetStatus "アンインストールシート作成" CreateFilteredSheet _ ws, _ wb, _ 10, _ Array( _ "6. アンインストール/書出し制限再設定", _ "6. アンインストール" _ ), _ "アンインストール" '================================================== ' 12. 書き出し制限解除 '========================================================== SetStatus "書き出し制限解除シート作成" CreateFilteredSheet _ ws, _ wb, _ 10, _ Array("5. 書出し制限解除"), _ "書き出し制限解除" '================================================== ' 13. 標準秘文 '========================================================== SetStatus "標準秘文シート作成" CreateFilteredSheet _ ws, _ wb, _ 10, _ Array( _ "1. 標準秘文（日）", _ "2. 標準秘文（英）" _ ), _ "標準秘文" '================================================== ' 14. 平文書出し秘文 '========================================================== SetStatus "平文書出し秘文シート作成" CreateFilteredSheet _ ws, _ wb, _ 10, _ Array("3. 平文書出し秘文"), _ "平文書出し秘文" '================================================== ' 15. オフライン秘文 '========================================================== SetStatus "オフライン秘文シート作成" CreateFilteredSheet _ ws, _ wb, _ 10, _ Array("4. オフライン秘文"), _ "オフライン秘文" '================================================== ' 各データシートのフィルターを3行目へ '========================================================== SetStatus "フィルターを再設定" For Each tmpWs In wb.Worksheets ResetFilter tmpWs Next tmpWs '================================================== ' サマリ作成 '========================================================== SetStatus "サマリシートを作成" Set summaryWs = CreateSummarySheet(wb) '================================================== ' 全シートA1選択 '========================================================== SetStatus "全シートをA1へ戻す" For Each tmpWs In wb.Worksheets tmpWs.Activate tmpWs.Range("A1").Select Next tmpWs ' 最後はサマリを表示 summaryWs.Activate summaryWs.Range("A1").Select '================================================== ' 完了 '========================================================== wb.Saved = False Application.CutCopyMode = False RestoreApplication _ oldCalc, _ oldScreen, _ oldEvents, _ oldAlerts, _ oldStatusBar MsgBox _ "作業用ブックを作成しました。" & vbCrLf & vbCrLf & _ "サマリシートも作成済みです。" & vbCrLf & _ "作成されたブックはまだ保存されていません。", _ vbInformation Exit Sub ErrorHandler: Dim errNo As Long Dim errText As String Dim errStep As String errNo = Err.Number errText = Err.Description errStep = gCurrentStep Application.CutCopyMode = False If stateSaved Then RestoreApplication _ oldCalc, _ oldScreen, _ oldEvents, _ oldAlerts, _ oldStatusBar End If MsgBox _ "処理中にエラーが発生しました。" & vbCrLf & vbCrLf & _ "処理：" & errStep & vbCrLf & _ "エラー番号：" & errNo & vbCrLf & _ "内容：" & errText & vbCrLf & vbCrLf & _ "途中まで作成されたブックは保存せず閉じてください。", _ vbCritical End Sub '========================================================== ' サマリシート作成 '========================================================== Private Function CreateSummarySheet( _ ByVal wb As Workbook _ ) As Worksheet Dim ws As Worksheet Dim unWs As Worksheet Set ws = wb.Worksheets.Add( _ Before:=wb.Worksheets(1)) ws.Name = "サマリ" ' 見出し ws.Range("A1").Value = "項目" ws.Range("B1").Value = "件数" ' 各シートのデータ行数 ws.Range("A2").Value = "標準秘文" ws.Range("B2").Value = _ CountDataRows(wb.Worksheets("標準秘文")) ws.Range("A3").Value = "平文書出し秘文" ws.Range("B3").Value = _ CountDataRows(wb.Worksheets("平文書出し秘文")) ws.Range("A4").Value = "オフライン秘文" ws.Range("B4").Value = _ CountDataRows(wb.Worksheets("オフライン秘文")) ' アンインストールの指定ステータス Set unWs = wb.Worksheets("アンインストール") ws.Range("A5").Value = _ "アンインストール：" & _ "ｱﾝｲﾝｽﾄｰﾙPW発行依頼連絡待ち" ws.Range("B5").Value = _ CountStatusRows( _ unWs, _ "ｱﾝｲﾝｽﾄｰﾙPW発行依頼連絡待ち" _ ) ' 見やすくする ws.Range("A1:B1").Font.Bold = True ws.Columns("A:B").AutoFit Set CreateSummarySheet = ws End Function '========================================================== ' データが入っている行数 '========================================================== Private Function CountDataRows( _ ByVal ws As Worksheet _ ) As Long Dim lastRow As Long Dim r As Long Dim cnt As Long lastRow = LastUsedRow(ws) If lastRow < FIRST_DATA_ROW Then CountDataRows = 0 Exit Function End If For r = FIRST_DATA_ROW To lastRow If Application.CountA( _ ws.Range( _ ws.Cells(r, 1), _ ws.Cells(r, 21) _ ) _ ) > 0 Then cnt = cnt + 1 End If Next r CountDataRows = cnt End Function '========================================================== ' D列の特定ステータス件数 '========================================================== Private Function CountStatusRows( _ ByVal ws As Worksheet, _ ByVal statusText As String _ ) As Long Dim lastRow As Long Dim r As Long Dim cnt As Long lastRow = LastUsedRow(ws) If lastRow < FIRST_DATA_ROW Then CountStatusRows = 0 Exit Function End If For r = FIRST_DATA_ROW To lastRow If StrComp( _ Trim$(CStr(ws.Cells(r, 4).Value2)), _ statusText, _ vbBinaryCompare _ ) = 0 Then cnt = cnt + 1 End If Next r CountStatusRows = cnt End Function '========================================================== ' 処理状況 '========================================================== Private Sub SetStatus(ByVal text As String) gCurrentStep = text Application.StatusBar = "処理中：" & text End Sub '========================================================== ' Excel状態復元 '========================================================== Private Sub RestoreApplication( _ ByVal oldCalc As XlCalculation, _ ByVal oldScreen As Boolean, _ ByVal oldEvents As Boolean, _ ByVal oldAlerts As Boolean, _ ByVal oldStatusBar As Variant) Application.Calculation = oldCalc Application.ScreenUpdating = oldScreen Application.EnableEvents = oldEvents Application.DisplayAlerts = oldAlerts Application.StatusBar = oldStatusBar End Sub '========================================================== ' 最終使用行 '========================================================== Private Function LastUsedRow( _ ByVal ws As Worksheet _ ) As Long Dim c As Range Set c = ws.Cells.Find( _ What:="*", _ After:=ws.Cells(1, 1), _ LookIn:=xlFormulas, _ LookAt:=xlPart, _ SearchOrder:=xlByRows, _ SearchDirection:=xlPrevious) If c Is Nothing Then LastUsedRow = FIRST_DATA_ROW - 1 Else LastUsedRow = c.Row End If End Function '========================================================== ' フィルター解除 '========================================================== Private Sub ClearFilter(ByVal ws As Worksheet) On Error Resume Next If ws.FilterMode Then ws.ShowAllData End If On Error GoTo 0 End Sub '========================================================== ' 3行目からフィルター '========================================================== Private Sub ResetFilter(ByVal ws As Worksheet) Dim lastRow As Long ClearFilter ws If ws.AutoFilterMode Then ws.AutoFilterMode = False End If lastRow = LastUsedRow(ws) If lastRow < FILTER_ROW Then lastRow = FILTER_ROW End If ws.Range( _ ws.Cells(FILTER_ROW, 1), _ ws.Cells(lastRow, 21) _ ).AutoFilter End Sub '========================================================== ' A列だけ値あり、B～Z空欄か '========================================================== Private Function IsItemOnlyRow( _ ByVal ws As Worksheet, _ ByVal rowNo As Long _ ) As Boolean If Len(Trim$(CStr(ws.Cells(rowNo, 1).Value2))) = 0 Then IsItemOnlyRow = False Exit Function End If IsItemOnlyRow = _ (Application.CountA( _ ws.Range( _ ws.Cells(rowNo, 2), _ ws.Cells(rowNo, 26) _ ) _ ) = 0) End Function '========================================================== ' S列空欄へ作業日 '========================================================== Private Sub FillWorkDate( _ ByVal ws As Worksheet, _ ByVal workDate As Date _ ) Dim lastRow As Long Dim r As Long lastRow = LastUsedRow(ws) If lastRow < FIRST_DATA_ROW Then Exit Sub For r = FIRST_DATA_ROW To lastRow If Len(CStr(ws.Cells(r, 19).Value2)) = 0 Then If Not IsItemOnlyRow(ws, r) Then ws.Cells(r, 19).Value = workDate End If End If Next r End Sub '========================================================== ' 項番のみの行削除 '========================================================== Private Sub DeleteItemOnlyRows(ByVal ws As Worksheet) Dim lastRow As Long Dim r As Long lastRow = LastUsedRow(ws) For r = lastRow To FIRST_DATA_ROW Step -1 If IsItemOnlyRow(ws, r) Then ws.Rows(r).Delete End If Next r End Sub '========================================================== ' 指定値に一致する / しない行を削除 '========================================================== Private Sub DeleteRowsMatching( _ ByVal ws As Worksheet, _ ByVal colNo As Long, _ ByVal values As Variant, _ ByVal deleteMatch As Boolean _ ) Dim lastRow As Long Dim arr As Variant Dim i As Long Dim rowNo As Long Dim matched As Boolean Dim doDelete As Boolean Dim blockTop As Long Dim blockBottom As Long lastRow = LastUsedRow(ws) If lastRow < FIRST_DATA_ROW Then Exit Sub arr = ws.Range( _ ws.Cells(FIRST_DATA_ROW, colNo), _ ws.Cells(lastRow, colNo) _ ).Value2 If lastRow = FIRST_DATA_ROW Then matched = ValueInList(arr, values) If deleteMatch Then doDelete = matched Else doDelete = Not matched End If If doDelete Then ws.Rows(FIRST_DATA_ROW).Delete End If Exit Sub End If For i = UBound(arr, 1) To 1 Step -1 rowNo = FIRST_DATA_ROW + i - 1 matched = ValueInList( _ arr(i, 1), _ values) If deleteMatch Then doDelete = matched Else doDelete = Not matched End If If doDelete Then If blockBottom = 0 Then blockTop = rowNo blockBottom = rowNo ElseIf rowNo = blockTop - 1 Then blockTop = rowNo Else ws.Rows( _ blockTop & ":" & blockBottom _ ).Delete blockTop = rowNo blockBottom = rowNo End If Else If blockBottom <> 0 Then ws.Rows( _ blockTop & ":" & blockBottom _ ).Delete blockTop = 0 blockBottom = 0 End If End If Next i If blockBottom <> 0 Then ws.Rows( _ blockTop & ":" & blockBottom _ ).Delete End If End Sub '========================================================== ' 指定値リスト内に存在するか '========================================================== Private Function ValueInList( _ ByVal target As Variant, _ ByVal values As Variant _ ) As Boolean Dim v As Variant Dim targetText As String If IsError(target) Then Exit Function targetText = Trim$(CStr(target)) For Each v In values If StrComp( _ targetText, _ Trim$(CStr(v)), _ vbBinaryCompare _ ) = 0 Then ValueInList = True Exit Function End If Next v End Function '========================================================== ' シートを丸ごと複製 → 対象行のみ残す '========================================================== Private Sub CreateFilteredSheet( _ ByVal srcWs As Worksheet, _ ByVal wb As Workbook, _ ByVal colNo As Long, _ ByVal values As Variant, _ ByVal newName As String _ ) Dim outWs As Worksheet gCurrentStep = newName & "：シートを複製" Application.StatusBar = "処理中：" & gCurrentStep srcWs.Copy _ After:=wb.Worksheets(wb.Worksheets.Count) Set outWs = _ wb.Worksheets(wb.Worksheets.Count) outWs.Name = newName gCurrentStep = newName & "：対象外データを削除" Application.StatusBar = "処理中：" & gCurrentStep ClearFilter outWs DeleteRowsMatching _ outWs, _ colNo, _ values, _ False ResetFilter outWs End Sub
