Option Explicit Private Const SRC_BOOK As String = "26年度申請台帳.xlsx" Private Const SRC_SHEET As String = "書出し設定申請" Private Const FILTER_ROW As Long = 3 Private Const FIRST_DATA_ROW As Long = 5 Sub 書出し設定申請_作業ブック作成() Dim srcWb As Workbook Dim srcWs As Worksheet Dim wb As Workbook Dim ws As Worksheet Dim lastRow As Long Dim workDate As Date Dim inputDate As String Dim oldCalc As XlCalculation Dim oldScreen As Boolean Dim oldEvents As Boolean Dim oldAlerts As Boolean '================================================== ' 作業日の入力 '================================================== Do inputDate = InputBox( _ Prompt:="作業日を入力してください。", _ Title:="作業日の入力", _ Default:=Format(Date, "yyyy/mm/dd")) If inputDate = "" Then Exit Sub If IsDate(inputDate) Then workDate = DateValue(CDate(inputDate)) Exit Do End If MsgBox "日付を正しく入力してください。", vbExclamation Loop '================================================== ' 元ブック確認 '================================================== On Error Resume Next Set srcWb = Workbooks(SRC_BOOK) On Error GoTo 0 If srcWb Is Nothing Then MsgBox _ SRC_BOOK & " を開いてから実行してください。", _ vbExclamation Exit Sub End If On Error Resume Next Set srcWs = srcWb.Worksheets(SRC_SHEET) On Error GoTo 0 If srcWs Is Nothing Then MsgBox _ "シート「" & SRC_SHEET & "」が見つかりません。", _ vbExclamation Exit Sub End If '================================================== ' Excel設定保存 '================================================== oldCalc = Application.Calculation oldScreen = Application.ScreenUpdating oldEvents = Application.EnableEvents oldAlerts = Application.DisplayAlerts Application.ScreenUpdating = False Application.EnableEvents = False Application.DisplayAlerts = False Application.Calculation = xlCalculationManual On Error GoTo ErrorHandler '================================================== ' 0. ' 書出し設定申請を新規ブックへコピー ' ' シート自体のコピーなので ' 条件付き書式等もそのまま引き継ぐ '================================================== srcWs.Copy Set wb = ActiveWorkbook Set ws = wb.Worksheets(1) ws.Name = SRC_SHEET ClearFilter ws '================================================== ' 2. ' D列：否認 を削除 '================================================== DeleteRowsMatching _ ws, _ 4, _ Array("否認"), _ True '================================================== ' 3. ' S列が空欄の申請へ作業日入力 ' ' 項番だけの行は除外 '================================================== FillWorkDate ws, workDate '================================================== ' 4. ' A列のみ値あり、 ' B～Z列がすべて空欄の項番行を削除 '================================================== DeleteItemOnlyRows ws '================================================== ' 5. ' N列：昇順 ' S列：新しい順 '================================================== lastRow = LastUsedRow(ws) If lastRow >= FIRST_DATA_ROW Then With ws.Sort .SortFields.Clear .SortFields.Add _ Key:=ws.Range( _ "N" & FIRST_DATA_ROW & _ ":N" & lastRow), _ SortOn:=xlSortOnValues, _ Order:=xlAscending, _ DataOption:=xlSortTextAsNumbers .SortFields.Add _ Key:=ws.Range( _ "S" & FIRST_DATA_ROW & _ ":S" & lastRow), _ SortOn:=xlSortOnValues, _ Order:=xlDescending, _ DataOption:=xlSortNormal .SetRange ws.Range( _ "A" & FIRST_DATA_ROW & _ ":Z" & lastRow) .Header = xlNo .MatchCase = False .Orientation = xlTopToBottom .Apply End With End If '================================================== ' 6. ' V～Z列削除 '================================================== ws.Columns("V:Z").Delete '================================================== ' 8. ' N列 = 0 / - ' 仮想端末シートへ ' ' シートを丸ごと複製してから ' 対象外の行を削除 '================================================== CreateFilteredSheet _ ws, _ wb, _ 14, _ Array("0", "-"), _ "仮想端末" '================================================== ' 9. ' 元シートから ' N列 = 0 / - を削除 '================================================== DeleteRowsMatching _ ws, _ 14, _ Array("0", "-"), _ True '================================================== ' 7・10 相当 ' ' N列の同一IT資産No.は ' 最初の1件だけ残す ' ' S列を新しい順に並べ替え済みなので ' 最初の1件 = 最新 '================================================== lastRow = LastUsedRow(ws) If lastRow >= FIRST_DATA_ROW Then ws.Range( _ ws.Cells(FIRST_DATA_ROW, 1), _ ws.Cells(lastRow, 21) _ ).RemoveDuplicates _ Columns:=14, _ Header:=xlNo End If '================================================== ' 11. ' アンインストール '================================================== CreateFilteredSheet _ ws, _ wb, _ 10, _ Array( _ "6. アンインストール/書出し制限再設定", _ "6. アンインストール" _ ), _ "アンインストール" '================================================== ' 12. ' 書き出し制限解除 '================================================== CreateFilteredSheet _ ws, _ wb, _ 10, _ Array( _ "5. 書出し制限解除" _ ), _ "書き出し制限解除" '================================================== ' 13. ' 標準秘文 '================================================== CreateFilteredSheet _ ws, _ wb, _ 10, _ Array( _ "1. 標準秘文（日）", _ "2. 標準秘文（英）" _ ), _ "標準秘文" '================================================== ' 14. ' 平文書き出し秘文 '================================================== CreateFilteredSheet _ ws, _ wb, _ 10, _ Array( _ "3. 平文書出し秘文" _ ), _ "平文書き出し秘文" '================================================== ' 15. ' オフライン秘文 '================================================== CreateFilteredSheet _ ws, _ wb, _ 10, _ Array( _ "4. オフライン秘文" _ ), _ "オフライン秘文" '================================================== ' 全シート ' 3行目にフィルター設定 '================================================== Dim tmpWs As Worksheet For Each tmpWs In wb.Worksheets ResetFilter tmpWs Next tmpWs '================================================== ' 完了 '================================================== ws.Activate ws.Range("A1").Select Application.CutCopyMode = False Application.Calculation = oldCalc Application.ScreenUpdating = oldScreen Application.EnableEvents = oldEvents Application.DisplayAlerts = oldAlerts MsgBox _ "作業用ブックを作成しました。" & vbCrLf & _ "条件付き書式等は元シートから引き継いでいます。" & vbCrLf & _ "新規ブックはまだ保存されていません。", _ vbInformation Exit Sub ErrorHandler: Application.CutCopyMode = False Application.Calculation = oldCalc Application.ScreenUpdating = oldScreen Application.EnableEvents = oldEvents Application.DisplayAlerts = oldAlerts MsgBox _ "処理中にエラーが発生しました。" & vbCrLf & vbCrLf & _ "エラー番号：" & Err.Number & vbCrLf & _ "内容：" & Err.Description, _ vbCritical End Sub '========================================================== ' 最終使用行 '========================================================== Private Function LastUsedRow( _ ByVal ws As Worksheet _ ) As Long Dim c As Range Set c = ws.Cells.Find( _ What:="*", _ After:=ws.Range("A1"), _ LookIn:=xlFormulas, _ LookAt:=xlPart, _ SearchOrder:=xlByRows, _ SearchDirection:=xlPrevious) If c Is Nothing Then LastUsedRow = FIRST_DATA_ROW - 1 Else LastUsedRow = c.Row End If End Function '========================================================== ' フィルター解除 '========================================================== Private Sub ClearFilter( _ ByVal ws As Worksheet _ ) On Error Resume Next If ws.FilterMode Then ws.ShowAllData End If On Error GoTo 0 End Sub '========================================================== ' フィルター再設定 '========================================================== Private Sub ResetFilter( _ ByVal ws As Worksheet _ ) Dim lastRow As Long ClearFilter ws If ws.AutoFilterMode Then ws.AutoFilterMode = False End If lastRow = LastUsedRow(ws) If lastRow < FILTER_ROW Then lastRow = FILTER_ROW End If ws.Range( _ ws.Cells(FILTER_ROW, 1), _ ws.Cells(lastRow, 21) _ ).AutoFilter End Sub '========================================================== ' 項番のみの行か ' ' A列：値あり ' B～Z：すべて空欄 '========================================================== Private Function IsItemOnlyRow( _ ByVal ws As Worksheet, _ ByVal rowNo As Long _ ) As Boolean If Len(Trim$(CStr(ws.Cells(rowNo, "A").Value))) = 0 Then IsItemOnlyRow = False Exit Function End If IsItemOnlyRow = _ (Application.CountA( _ ws.Range( _ "B" & rowNo & _ ":Z" & rowNo _ ) _ ) = 0) End Function '========================================================== ' S列空欄へ作業日 '========================================================== Private Sub FillWorkDate( _ ByVal ws As Worksheet, _ ByVal workDate As Date _ ) Dim lastRow As Long Dim r As Long lastRow = LastUsedRow(ws) For r = FIRST_DATA_ROW To lastRow If Len(ws.Cells(r, "S").Value) = 0 Then If Not IsItemOnlyRow(ws, r) Then ws.Cells(r, "S").Value = workDate ws.Cells(r, "S").NumberFormatLocal = _ "yyyy/mm/dd" End If End If Next r End Sub '========================================================== ' 項番だけの行削除 '========================================================== Private Sub DeleteItemOnlyRows( _ ByVal ws As Worksheet _ ) Dim lastRow As Long Dim r As Long lastRow = LastUsedRow(ws) For r = lastRow To FIRST_DATA_ROW Step -1 If IsItemOnlyRow(ws, r) Then ws.Rows(r).Delete End If Next r End Sub '========================================================== ' 指定値に一致する / 一致しない行をまとめて削除 ' ' deleteMatch = True ' → valuesに一致する行を削除 ' ' deleteMatch = False ' → valuesに一致しない行を削除 '========================================================== Private Sub DeleteRowsMatching( _ ByVal ws As Worksheet, _ ByVal colNo As Long, _ ByVal values As Variant, _ ByVal deleteMatch As Boolean _ ) Dim lastRow As Long Dim arr As Variant Dim i As Long Dim rowNo As Long Dim matched As Boolean Dim doDelete As Boolean Dim blockTop As Long Dim blockBottom As Long lastRow = LastUsedRow(ws) If lastRow < FIRST_DATA_ROW Then Exit Sub arr = ws.Range( _ ws.Cells(FIRST_DATA_ROW, colNo), _ ws.Cells(lastRow, colNo) _ ).Value2 For i = UBound(arr, 1) To 1 Step -1 rowNo = FIRST_DATA_ROW + i - 1 matched = ValueInList( _ arr(i, 1), _ values) If deleteMatch Then doDelete = matched Else doDelete = Not matched End If If doDelete Then If blockBottom = 0 Then blockTop = rowNo blockBottom = rowNo ElseIf rowNo = blockTop - 1 Then blockTop = rowNo Else ws.Rows( _ blockTop & ":" & blockBottom _ ).Delete blockTop = rowNo blockBottom = rowNo End If Else If blockBottom <> 0 Then ws.Rows( _ blockTop & ":" & blockBottom _ ).Delete blockTop = 0 blockBottom = 0 End If End If Next i If blockBottom <> 0 Then ws.Rows( _ blockTop & ":" & blockBottom _ ).Delete End If End Sub '========================================================== ' 値が指定配列に含まれるか '========================================================== Private Function ValueInList( _ ByVal target As Variant, _ ByVal values As Variant _ ) As Boolean Dim v As Variant Dim targetText As String If IsError(target) Then ValueInList = False Exit Function End If targetText = Trim$(CStr(target)) For Each v In values If StrComp( _ targetText, _ Trim$(CStr(v)), _ vbBinaryCompare _ ) = 0 Then ValueInList = True Exit Function End If Next v ValueInList = False End Function '========================================================== ' シート丸ごと複製 → 対象行だけ残す ' ' Range.Copyを使わないため ' 条件付き書式等も保持 '========================================================== Private Sub CreateFilteredSheet( _ ByVal srcWs As Worksheet, _ ByVal wb As Workbook, _ ByVal colNo As Long, _ ByVal values As Variant, _ ByVal newName As String _ ) Dim outWs As Worksheet ' シートそのものを複製 srcWs.Copy _ After:=wb.Worksheets( _ wb.Worksheets.Count) Set outWs = _ wb.Worksheets( _ wb.Worksheets.Count) outWs.Name = newName ClearFilter outWs ' 指定値以外を削除 DeleteRowsMatching _ outWs, _ colNo, _ values, _ False ResetFilter outWs End Sub
