function Invoke-FixedCapture { if (-not $script:CaptureRect -or $script:CaptureRect.Width -le 0 -or $script:CaptureRect.Height -le 0) { Show-Info "未設定" "先に $($script:HotkeySetRect.Text) で範囲を設定してください。" 2000 return } $rect = $script:CaptureRect try { # ------------------------------------------------- # 1. 固定範囲をキャプチャ # ------------------------------------------------- $bmp = New-Object System.Drawing.Bitmap($rect.Width, $rect.Height) $g = [System.Drawing.Graphics]::FromImage($bmp) $g.CopyFromScreen( (New-Object System.Drawing.Point($rect.X, $rect.Y)), [System.Drawing.Point]::Empty, (New-Object System.Drawing.Size($rect.Width, $rect.Height)) ) # ------------------------------------------------- # 2. 黒色 2px の境界線を追加 # ------------------------------------------------- $borderSize = 2 $brush = New-Object System.Drawing.SolidBrush([System.Drawing.Color]::Black) # 上 $g.FillRectangle( $brush, 0, 0, $bmp.Width, $borderSize ) # 下 $g.FillRectangle( $brush, 0, $bmp.Height - $borderSize, $bmp.Width, $borderSize ) # 左 $g.FillRectangle( $brush, 0, 0, $borderSize, $bmp.Height ) # 右 $g.FillRectangle( $brush, $bmp.Width - $borderSize, 0, $borderSize, $bmp.Height ) $brush.Dispose() $g.Dispose() # ------------------------------------------------- # 3. 横幅780pxまでアスペクト比を維持して拡大 # ------------------------------------------------- $targetWidth = 780 if ($bmp.Width -lt $targetWidth) { $scale = $targetWidth / [double]$bmp.Width $targetHeight = [int][Math]::Round($bmp.Height * $scale) $resized = New-Object System.Drawing.Bitmap( $targetWidth, $targetHeight ) $rg = [System.Drawing.Graphics]::FromImage($resized) # スクリーンショット向けの高品質拡大 $rg.InterpolationMode = ` [System.Drawing.Drawing2D.InterpolationMode]::HighQualityBicubic $rg.PixelOffsetMode = ` [System.Drawing.Drawing2D.PixelOffsetMode]::HighQuality $rg.CompositingQuality = ` [System.Drawing.Drawing2D.CompositingQuality]::HighQuality $rg.DrawImage( $bmp, 0, 0, $targetWidth, $targetHeight ) $rg.Dispose() $bmp.Dispose() $bmp = $resized } # ------------------------------------------------- # 4. クリップボードへ # ------------------------------------------------- if (Set-ClipboardImageRetry -Image $bmp) { Show-Info "キャプチャ完了" ` ("{0} x {1} をクリップボードへコピーしました。" -f ` $bmp.Width, $bmp.Height) 1200 [System.Media.SystemSounds]::Asterisk.Play() } else { Show-Info "失敗" "クリップボードへの格納に失敗しました。" 2000 } $bmp.Dispose() } catch { Show-Info "エラー" $_.Exception.Message 2500 } }
