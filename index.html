SetStatus "元シートを新規ブックへコピー" srcWs.Copy Set wb = ActiveWorkbook Set ws = wb.Worksheets(1) ws.Name = SRC_SHEET ClearFilter ws を置き換え ＝＝＝＝＝＝＝＝＝＝＝＝＝＝＝＝＝ SetStatus "元ブックから作業用コピーを作成" Dim tempPath As String Dim tempName As String Dim delWs As Worksheet ' 一時ファイル名 tempName = _ "書出し設定申請_作業用_" & _ Format(Now, "yyyymmdd_hhnnss") & ".xlsx" tempPath = Environ$("TEMP") & "\" & tempName ' 元ブック全体を複製 srcWb.SaveCopyAs tempPath ' 複製したブックを開く Set wb = Workbooks.Open( _ Filename:=tempPath, _ UpdateLinks:=False, _ ReadOnly:=False) Set ws = wb.Worksheets(SRC_SHEET) '================================================== ' 書出し設定申請以外のシートを削除 '================================================== SetStatus "不要シートを削除" Application.DisplayAlerts = False For Each delWs In wb.Worksheets If delWs.Name <> SRC_SHEET Then delWs.Delete End If Next delWs Application.DisplayAlerts = False ClearFilter ws ＝＝＝＝＝＝ ws.Activate ws.Range("A1").Select の直後に ＝＝＝＝ ' 一時コピーから作ったブックなので ' ユーザーに保存を促す状態にする wb.Saved = False
